function modelYukle(url, ikinciDeneme = false) { document.getElementById('status').innerText = "Model yükleniyor..."; loader.load(url, (gltf) => { scene.remove(loadedModel); loadedModel = gltf.scene; // 1. Model materyallerini kontrol et ve görünür yap loadedModel.traverse((child) => { if (child.isMesh) { child.material.side = THREE.DoubleSide; // Çift taraflı kaplama child.material.frustumCulled = false; // Kameradan kaçmasını engelle } }); // 2. Modelin boyutunu ve merkezini hesapla const box = new THREE.Box3().setFromObject(loadedModel); const center = box.getCenter(new THREE.Vector3()); const size = box.getSize(new THREE.Vector3()); // Modeli orijine (0,0,0) taşı loadedModel.position.x = -center.x; loadedModel.position.y = -center.y; loadedModel.position.z = -center.z; scene.add(loadedModel); // 3. Kamerayı modelin boyutuna göre dinamik hizala const maxDim = Math.max(size.x, size.y, size.z); const fov = camera.fov * (Math.PI / 180); let cameraZ = Math.abs(maxDim / 2 / Math.tan(fov / 2)) * 2.5; // Kamera çok yakın/uzak kalmasın diye sınır koy cameraZ = Math.max(cameraZ, 2); camera.position.set(0, maxDim * 0.3, cameraZ); camera.lookAt(0, 0, 0); camera.near = cameraZ / 100; camera.far = cameraZ * 100; camera.updateProjectionMatrix(); controls.target.set(0, 0, 0); controls.update(); document.getElementById('status').innerText = "✅ Model başarıyla yüklendi!"; }, undefined, (error) => { console.error("Yükleme hatası:", error); if (!ikinciDeneme) { const tamUrl = window.location.href.substring(0, window.location.href.lastIndexOf('/') + 1) + 'motor10.glb'; modelYukle(tamUrl, true); } else { document.getElementById('status').innerText = "❌ Model yükleme hatası! Dosya adı 'motor.glb' olmalı."; } }); }
